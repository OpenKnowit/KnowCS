### Tensor.addcmul Tensor.addcdiv Tensor.addcdiv_
import torch
t = torch.tensor([1., 2., 3.])
a = torch.tensor([2., 4., 6.])
b = torch.tensor([4., 2., 1.])
print(t.addcmul(a, b, value=0.5))     # t + 0.5 * a * b
print(t.addcdiv(a, b, value=0.5))     # t + 0.5 * a / b
t.addcdiv_(a, b)                      # the same with value=1, in place
print(t)

### torch.addmm torch.addmv torch.addr Tensor.addmm_ Tensor.addmv_ Tensor.addr_ torch.addmv_
import torch
M = torch.zeros(2, 2)
A = torch.tensor([[1., 2.], [3., 4.]])
v = torch.tensor([1., 1.])
w = torch.tensor([1., 2.])
print(torch.addmm(M, A, A, beta=1, alpha=2))     # beta * M + alpha * (A @ A)
print(torch.addmv(v, A, v))                      # v + A @ v
print(torch.addr(M, v, w))                       # M + outer(v, w)
M.addmm_(A, torch.eye(2))                        # in place: M += A @ I
M.addr_(v, w)
print(M)
u = torch.ones(2)
u.addmv_(A, v)
torch.addmv_(u, A, v)
print(u)

### Tensor.bmm torch.baddbmm torch.addbmm Tensor.baddbmm_ Tensor.addbmm_
import torch
A = torch.tensor([[1., 2.], [3., 4.]])

print(batch.bmm(batch))                           # each matrix times itself

C = torch.zeros(2, 2, 2)
C.baddbmm_(batch, batch)
S = torch.zeros(2, 2)
S.addbmm_(batch, batch)
print(C[1], S)

### Tensor.cholesky Tensor.qr Tensor.solve Tensor.eig Tensor.symeig Tensor.lstsq
import torch
A = torch.tensor([[4., 2.], [2., 3.]])
# the old Tensor methods were removed in torch 2; torch.linalg has them now
for name, call in [('cholesky', lambda: A.cholesky()), ('qr', lambda: A.qr()), ('solve', lambda: A.solve(A)),
                   ('eig', lambda: A.eig()), ('symeig', lambda: A.symeig()), ('lstsq', lambda: A.lstsq(A))]:
    try:
        call()
        print(name, 'works')
    except RuntimeError:
        print(name, 'removed')
print(torch.linalg.cholesky(A))                    # A = L @ L.T
print(torch.linalg.solve(A, torch.tensor([2., 1.])))
