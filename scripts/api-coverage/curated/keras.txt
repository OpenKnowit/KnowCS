### keras.Layer keras.Model.compute_output_shape
import numpy as np
import keras

class Scale(keras.Layer):                         # a layer of your own: one weight per feature
    def build(self, input_shape):
        self.w = self.add_weight(shape=(input_shape[-1],), initializer='ones')
    def call(self, x):
        return x * self.w

model = keras.Sequential([keras.Input((3,)), Scale(), keras.layers.Dense(2)])
model.summary()
print(model.compute_output_shape((None, 3)))

### keras.losses.Loss keras.losses.sparse_categorical_crossentropy keras.metrics.sparse_categorical_crossentropy
import numpy as np
import keras

class MeanAbs(keras.losses.Loss):                 # a loss of your own
    def call(self, y_true, y_pred):
        return keras.ops.mean(keras.ops.abs(y_true - y_pred), axis=-1)

print(MeanAbs()(np.array([[1., 2.]]), np.array([[1.5, 1.]])))
probs = np.array([[0.7, 0.2, 0.1], [0.1, 0.1, 0.8]])
labels = np.array([0, 2])                          # class numbers, not one-hot
print(keras.losses.sparse_categorical_crossentropy(labels, probs))   # −log 0.7, −log 0.8
print(keras.metrics.sparse_categorical_crossentropy(labels, probs))

### keras.metrics.Metric keras.metrics.sparse_categorical_accuracy keras.metrics.sparse_top_k_categorical_accuracy keras.metrics.get
import numpy as np
import keras
probs = np.array([[0.7, 0.2, 0.1], [0.3, 0.6, 0.1], [0.5, 0.1, 0.4]], dtype='float32')
labels = np.array([0, 0, 2])
print(keras.metrics.sparse_categorical_accuracy(labels, probs))         # 1 where the top class is right
print(keras.metrics.sparse_top_k_categorical_accuracy(labels, probs, k=2))
acc = keras.metrics.get('accuracy')
print(isinstance(keras.metrics.SparseCategoricalAccuracy(), keras.metrics.Metric), callable(acc))

### keras.activations.get keras.activations.serialize keras.activations.deserialize keras.initializers.serialize keras.regularizers.get
import numpy as np
import keras
relu = keras.activations.get('relu')               # a name → the function

print(keras.activations.serialize(relu))           # and back to its name
print(keras.activations.deserialize('sigmoid')(np.array([0.])))
print(keras.initializers.serialize(keras.initializers.Zeros()))

### keras.ops.tensordot
import numpy as np
import keras
a = np.arange(6.).reshape(2, 3)
b = np.arange(12.).reshape(3, 4)
print(keras.ops.tensordot(a, b, axes=1))           # the same as a @ b
print(keras.ops.tensordot(a, a, axes=2))           # every product summed: a scalar

### keras.layers.Input keras.layers.Conv2D keras.optimizers.Adam keras.optimizers.get keras.activations.tanh keras.utils.to_categorical
import numpy as np
import keras
from keras import layers
model = keras.Sequential([
    layers.Input((8, 8, 1)),                         # 8 × 8 grey pictures
    layers.Conv2D(4, 3, activation='relu'),          # 4 filters of 3 × 3 → (6, 6, 4)
    layers.Flatten(),
    layers.Dense(3, activation='softmax'),
])
model.summary()
print(keras.utils.to_categorical([0, 2, 1], num_classes=3))      # labels → one-hot rows
opt = keras.optimizers.Adam(learning_rate=0.01)
print(type(opt).__name__, float(opt.learning_rate), type(keras.optimizers.get('sgd')).__name__)
print(keras.activations.tanh(np.array([-2., 0., 2.])))

### keras.Model.to_json keras.Model.get_compile_config
import numpy as np
import keras
from keras import layers
model = keras.Sequential([layers.Input((2,)), layers.Dense(1)])
text = model.to_json()                      # the architecture as JSON, without the weights
again = keras.models.model_from_json(text)
again.summary()
model.compile(optimizer='sgd', loss='mse')
print(model.get_compile_config()['loss'])
