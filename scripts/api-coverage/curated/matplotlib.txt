### Axes.get_xlim Axes.get_xbound Axes.get_ybound Axes.get_xmargin Axes.get_ymargin Axes.viewLim Axes.get_data_ratio

import matplotlib.pyplot as plt

### plt.MultipleLocator plt.MaxNLocator plt.FixedLocator plt.LinearLocator plt.NullLocator plt.AutoLocator plt.LogLocator
import numpy as np
import matplotlib.pyplot as plt
x = np.linspace(0, 10, 50)
fig, axs = plt.subplots(2, 3, figsize=(9, 4))
for a in axs.flat:
    a.plot(x, x)
axs[0, 0].xaxis.set_major_locator(plt.MultipleLocator(2.5))    # every 2.5
axs[0, 1].xaxis.set_major_locator(plt.MaxNLocator(3))          # at most 3 intervals
axs[0, 2].xaxis.set_major_locator(plt.FixedLocator([1, 4, 9]))
axs[1, 0].xaxis.set_major_locator(plt.LinearLocator(4))        # 4 evenly spaced
axs[1, 1].xaxis.set_major_locator(plt.NullLocator())           # none
axs[1, 2].xaxis.set_major_locator(plt.AutoLocator())
axs[1, 2].set_yscale('log')
axs[1, 2].yaxis.set_major_locator(plt.LogLocator())
