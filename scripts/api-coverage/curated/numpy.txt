# Programs for the API index of NumPy names whose automatic example only fetched them, without a call.

### ndarray.argsort ndarray.argpartition ndarray.partition ndarray.nonzero
import numpy as np
x = np.array([30, 10, 50, 20, 40])
print(x.argsort())             # positions that would sort x: smallest first
print(x[x.argsort()])
y = x.copy()
y.partition(2)                 # the 3rd smallest in place 2, smaller ones before it (in any order)
print(y[2], x.argpartition(2)[2])
print((x > 25).nonzero())      # where the condition holds

### ndarray.flatten ndarray.squeeze ndarray.swapaxes ndarray.transpose ndarray.dot
import numpy as np
x = np.arange(6).reshape(1, 2, 3)
print(x.squeeze().shape)       # the length-1 axis dropped
print(x.flatten())             # a 1-D copy
print(x.swapaxes(1, 2).shape, x.transpose(2, 1, 0).shape)
m = np.array([[1, 2], [3, 4]])
print(m.transpose())
print(m.dot(np.array([1, 1])))   # matrix times vector: row sums

### np.frompyfunc np.promote_types np.typename np.get_printoptions
import numpy as np
add_twice = np.frompyfunc(lambda a, b: a + 2 * b, 2, 1)    # a Python function made elementwise
print(list(add_twice(np.array([1, 2, 3]), 10)))
print(np.promote_types(np.int64, np.float32), np.promote_types('i4', 'f2'))   # the smallest type both fit in
print(np.typename('f'), np.typename('i'))
print(np.get_printoptions()['precision'])

### np.genfromtxt
import numpy as np
import io
text = io.StringIO("x,y\n1,2.5\n3,\n5,6.0")
data = np.genfromtxt(text, delimiter=',', skip_header=1)    # a missing value becomes nan
print(data)
print(np.isnan(data))                                       # where the gaps were

### np.random.Generator np.random.get_state np.random.triangular np.random.negative_binomial
import numpy as np
rng = np.random.default_rng(0)
print(isinstance(rng, np.random.Generator), rng.integers(0, 10, 3).shape)   # the recommended interface
np.random.seed(0)
state = np.random.get_state()             # save the legacy generator
a = np.random.triangular(0, 1, 4, size=3) # most values near the peak, 1
np.random.set_state(state)
b = np.random.triangular(0, 1, 4, size=3)
print(np.array_equal(a, b), a.round(3))
print(np.random.negative_binomial(3, 0.5, size=5))   # failures before the 3rd success

### np.transpose np.moveaxis ndarray.var ndarray.strides
import numpy as np
images = np.arange(24).reshape(2, 3, 4)          # 2 pictures, 3 rows, 4 columns
print(np.transpose(images, (0, 2, 1)).shape)    # rows and columns swapped in each
print(np.moveaxis(images, 0, -1).shape)         # the first axis moved to the end
m = np.array([[1., 2.], [3., 4.]])
print(np.transpose(m))
print(m.var(), m.var(axis=0))                   # spread of all values, then of each column
print(m.strides)                                # bytes to step along each axis: 2 × 8, 8
