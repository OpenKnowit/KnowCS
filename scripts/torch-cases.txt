import torch
import torch.nn as nn
import torch.nn.functional as F
class Net(nn.Module):
    def __init__(self, n=10):
        super().__init__()
        self.conv1 = nn.Conv2d(1, 8, 3, padding=1)
        self.pool = nn.MaxPool2d(2)
        self.conv2 = nn.Conv2d(8, 16, kernel_size=3, stride=1)
        self.bn = nn.BatchNorm2d(16)
        self.drop = nn.Dropout(0.25)
        self.fc1 = nn.Linear(16 * 6 * 6, 32)
        self.fc2 = nn.Linear(32, n)
    def forward(self, x):
        x = self.pool(F.relu(self.conv1(x)))
        x = self.pool(F.relu(self.bn(self.conv2(x))))
        x = torch.flatten(x, 1)
        x = self.drop(F.relu(self.fc1(x)))
        return self.fc2(x)
m = Net()

class Block(nn.Module):
    def __init__(self, d):
        super().__init__()
        self.ls = nn.ModuleList([nn.Linear(d, d) for _ in range(3)])

# training with a validation loop
<<<
import torch
import torch.nn as nn
from torch.utils.data import DataLoader, TensorDataset
X = torch.arange(20.).view(10, 2) / 10
y = (X.sum(1) > 1).long()
tr = DataLoader(TensorDataset(X, y), batch_size=4)
m = nn.Sequential(nn.Flatten(), nn.Linear(2, 3), nn.ReLU(), nn.Linear(3, 2))
with torch.no_grad():
    for k, p in enumerate(m.parameters()):
        p.copy_(((torch.arange(p.numel()) * 3 + k) % 7 - 3).float().view(p.shape) / 5)
opt = torch.optim.Adam(m.parameters(), lr=0.05, weight_decay=1e-3)
sch = torch.optim.lr_scheduler.StepLR(opt, step_size=2, gamma=0.5)
crit = nn.CrossEntropyLoss()
for ep in range(4):
    m.train()
    tot = 0.0
    for xb, yb in tr:
        opt.zero_grad()
        out = m(xb)
        l = crit(out, yb)
        l.backward()
        opt.step()
        tot += l.item() * xb.size(0)
    sch.step()
    m.eval()
    with torch.no_grad():
        pred = m(X).argmax(dim=1)
        acc = (pred == y).float().mean().item()
    print(f'epoch {ep}: loss {tot / len(X):.4f} acc {acc:.3f} lr {sch.get_last_lr()[0]:.4f}')
